357: 
358: /// Atomically reads a WorkspaceDocument and inserts a new VaultItem from its content.
359: ///
360: /// The original WorkspaceDocument row is left completely untouched — Save-to-Vault
361: /// is a copy/promotion, never a move or delete.
362: ///
363: /// Edge case (documented, not resolved here): the source tab's CHANGE 08 dot indicator
364: /// will remain visible after promotion because it is driven by the WorkspaceDocument
365: /// TypeScript type, not a "promoted" flag. This is an accepted known limitation per the
366: /// CHANGE 10 spec — resolving dot semantics for already-promoted workspace tabs is deferred.
367: #[tauri::command]
368: fn save_workspace_document_to_vault(
369:     id: String, // WorkspaceDocument id to promote
370:     state: tauri::State<db::DbState>,
371: ) -> Result<(), String> {
372:     use chrono::Utc;
373:     use uuid::Uuid;
374: 
375:     let mut guard = state.0.lock().map_err(|e| e.to_string())?;
376:     let tx = guard
377:         .transaction()
378:         .map_err(|e| format!("save_workspace_document_to_vault transaction: {}", e))?;
379: 
380:     // 1. Read source WorkspaceDocument — fail fast if id is not found.
381:     let (title, body, type_str, source_vault_item_id, doc_url): (String, String, String, Option<String>, Option<String>) = tx
382:         .query_row(
383:             "SELECT title, body, suggested_type, source_vault_item_id, url FROM workspace_documents WHERE id = ?1",
384:             rusqlite::params![id],
385:             |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?, row.get(3)?, row.get(4)?)),
386:         )
387:         .map_err(|e| format!("save_workspace_document_to_vault read: {}", e))?;
388: 
389:     // 2. Map suggested_type → committed vault type.
390:     //    Image / File are reserved-future types — fall back to Note.
391:     let vault_type_str = match models::ContentType::from_db_str(&type_str)
392:         .unwrap_or(models::ContentType::Note)
393:     {
394:         models::ContentType::CodeSnippet => "codeSnippet",
395:         models::ContentType::Link        => "link",
396:         _                                => "note",
397:     };
398: 
399:     // 3. Insert new VaultItem.
400:     //    source_url / source_domain / source_title / captured_at → all NULL
401:     //    (workspace docs have no URL provenance metadata).
402:     //    vault_item_tags → empty (promotion-time tag entry deferred to CHANGE 11+).
403:     // 3. Read tags from workspace_document_tags
404:     let mut tag_stmt = tx.prepare("SELECT tag FROM workspace_document_tags WHERE doc_id = ?1").unwrap();
405:     let tag_rows = tag_stmt.query_map(rusqlite::params![id], |row| row.get::<_, String>(0)).unwrap();
406:     let mut tags = Vec::new();
407:     for t in tag_rows {
408:         if let Ok(tag) = t {
409:             tags.push(tag);
410:         }
411:     }
412:     drop(tag_stmt); // Drop before borrowing tx for execute
413: 
414:     // 4. Insert or Update VaultItem.
415:     let now = Utc::now().to_rfc3339();
416:     
417:     let (link_source_url, link_source_domain) = if vault_type_str == "link" {
418:         let u_opt = doc_url.as_ref().map(|s| s.trim()).filter(|s| !s.is_empty());
419:         let domain = u_opt.and_then(|trimmed| {
420:             let without_proto = if let Some(stripped) = trimmed.strip_prefix("https://") {
421:                 stripped
422:             } else if let Some(stripped) = trimmed.strip_prefix("http://") {
423:                 stripped
424:             } else {
425:                 trimmed
426:             };
427:             let host = without_proto.split('/').next().unwrap_or(without_proto);
428:             let host = host.split(':').next().unwrap_or(host);
429:             if host.is_empty() { None } else { Some(host.to_string()) }
430:         });
431:         (u_opt.map(|s| s.to_string()), domain)
432:     } else {
433:         (None, None)
434:     };
435: 
436:     let target_vault_id = if let Some(vault_id) = source_vault_item_id {
437:         tx.execute(
438:             "UPDATE vault_items SET title = ?1, body = ?2, type = ?3, updated_at = ?4, source_url = COALESCE(?5, source_url), source_domain = COALESCE(?6, source_domain) WHERE id = ?7",
439:             rusqlite::params![title, body, vault_type_str, now, link_source_url, link_source_domain, vault_id.clone()],
440:         )
441:         .map_err(|e| format!("save_workspace_document_to_vault update: {}", e))?;
442:         vault_id
443:     } else {
444:         let vault_id = Uuid::new_v4().to_string();
445:         tx.execute(
446:             "INSERT INTO vault_items
447:              (id, title, body, type, source_url, source_domain,
448:               source_title, captured_at, created_at, updated_at)
449:              VALUES (?1, ?2, ?3, ?4, ?5, ?6, NULL, NULL, ?7, ?8)",
450:             rusqlite::params![vault_id, title, body, vault_type_str, link_source_url, link_source_domain, now.clone(), now],
451:         )
452:         .map_err(|e| format!("save_workspace_document_to_vault insert: {}", e))?;
453:         vault_id
454:     };
455: 
456:     // 5. Update vault_item_tags
457:     tx.execute("DELETE FROM vault_item_tags WHERE item_id = ?1", rusqlite::params![target_vault_id])
458:         .map_err(|e| format!("save_workspace_document_to_vault delete tags: {}", e))?;
459:     
460:     for tag in tags {
461:         tx.execute(
462:             "INSERT INTO vault_item_tags (item_id, tag) VALUES (?1, ?2)",
463:             rusqlite::params![target_vault_id, tag],
464:         )
465:         .map_err(|e| format!("save_workspace_document_to_vault insert tags: {}", e))?;
466:     }
467: 
468:     tx.commit()
469:         .map_err(|e| format!("save_workspace_document_to_vault commit: {}", e))?;
470: 
471:     Ok(())
472: }

541: // ── CHANGE 13: Clipboard service ─────────────────────────────────────────
542: 
543: /// Discriminated read result — matches the TypeScript ClipboardReadResult union.
544: /// `serde(tag = "kind")` produces { "kind": "text", "content": "..." } etc.
545: #[derive(Debug, Clone, serde::Serialize)]
546: #[serde(tag = "kind", rename_all = "camelCase")]
547: enum ClipboardReadResult {
548:     Text { content: String },
549:     Image { width: usize, height: usize },
550:     Unsupported, // clipboard system inaccessible (locked, init failure)
551:     Empty,       // clipboard accessible but nothing we can read
552: }
553: 
554: impl ClipboardReadResult {
555:     fn kind_label(&self) -> &'static str {
556:         match self {
557:             ClipboardReadResult::Text { .. } => "text",
558:             ClipboardReadResult::Image { .. } => "image",
559:             ClipboardReadResult::Unsupported => "unsupported",
560:             ClipboardReadResult::Empty => "empty",
561:         }
562:     }
563: }
564: 
565: /// Internal helper for reading clipboard content via arboard.
566: fn read_clipboard_inner() -> ClipboardReadResult {
567:     use arboard::Clipboard;
568: 
569:     // Opening clipboard can fail transiently (e.g. another process holds the lock).
570:     let mut cb = match Clipboard::new() {
571:         Ok(c) => c,
572:         Err(_) => return ClipboardReadResult::Unsupported,
573:     };
574: 
575:     // 1. Try plain text (arboard returns ContentNotAvailable if format absent).
576:     match cb.get_text() {
577:         Ok(text) if !text.trim().is_empty() => {
578:             return ClipboardReadResult::Text { content: text };
579:         }
580:         Ok(_) => {} // empty string — fall through
581:         Err(arboard::Error::ContentNotAvailable) => {} // no text — fall through
582:         Err(_) => {} // other error — treat as fall-through, try image next
583:     }
584: 
585:     // 2. Try image (e.g. copied via Win+Shift+S or a browser "copy image").
586:     //    Raw RGBA bytes are read but not forwarded — the frontend only needs
587:     //    the dimensions for CHANGE 14 type detection.
588:     //    Persisting images as Vault attachments is out of scope (v1 spec).
589:     match cb.get_image() {
590:         Ok(img) if img.width > 0 && img.height > 0 => {
591:             return ClipboardReadResult::Image {
592:                 width: img.width,
593:                 height: img.height,
594:             };
595:         }
596:         Ok(_) => {} // zero-size image — fall through
597:         Err(arboard::Error::ContentNotAvailable) => {} // no image — fall through
598:         Err(_) => {} // other error — fall through
599:     }
600: 
601:     // 3. Clipboard accessible but nothing readable.
602:     ClipboardReadResult::Empty
603: }
604: 
605: /// Reads the current Windows clipboard content.
606: /// Tries text first (most common), then image, then signals empty/unsupported.
607: /// Never panics — all errors are mapped to Unsupported or Empty variants.
608: #[tauri::command]
609: fn read_clipboard() -> ClipboardReadResult {
610:     read_clipboard_inner()
611: }

616: /// Converts the stored Hotkey struct to the format Tauri's global-shortcut
617: /// plugin expects: "Ctrl+Shift+S", "Alt+F", "Ctrl+Shift+F1", etc.
618: fn hotkey_to_shortcut_str(hotkey: &models::Hotkey) -> String {
619:     let mut parts: Vec<String> = hotkey
620:         .modifiers
621:         .iter()
622:         .map(|m| match m.to_lowercase().as_str() {
623:             "ctrl" => "Ctrl".to_string(),
624:             "shift" => "Shift".to_string(),
625:             "alt" => "Alt".to_string(),
626:             "meta" => "Super".to_string(), // Windows key
627:             other => {
628:                 let mut s = other.to_string();
629:                 if let Some(c) = s.get_mut(0..1) {
630:                     c.make_ascii_uppercase();
631:                 }
632:                 s
633:             }
634:         })
635:         .collect();
636:     // Primary key: single chars capitalized; named keys (f1, space, etc.) left as-is
637:     let key = if hotkey.key.len() == 1 {
638:         hotkey.key.to_uppercase()
639:     } else {
640:         hotkey.key.clone()
641:     };
642:     parts.push(key);
643:     parts.join("+")
644: }

441:     }
442:

444:     for doc in &mut docs {

452:         for t in tag_rows {

454:         }
455:         doc.tags = tags;
456:     }

458:     Ok(docs)
